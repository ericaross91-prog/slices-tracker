⚙ Connect Google Sheets

STEP 1 — Go to sheets.google.com and create a new blank spreadsheet. Name it SLICES Tracker Data.

STEP 2 — Inside that sheet click Extensions → Apps Script. Delete all default code.

STEP 3 — Click the button below to copy the code, paste it into the editor, and save with Ctrl+S.

STEP 4 — Click Deploy → New deployment → gear icon → Web app. Set Execute as: Me and Who has access: Anyone. Click Deploy and authorize.

STEP 5 — Copy the Web app URL and paste it below. It starts with https://script.google.com/macros/s/
Apps Script code:
const SHEET_NAME = 'TrackerData'; function doPost(e) { try { const raw = e.postData && e.postData.contents ? e.postData.contents : (e.parameter && e.parameter.payload ? e.parameter.payload : '{}'); const data = JSON.parse(raw); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); if (data.action === 'write' && data.rows) { sheet.clearContents(); const h = ['id','business','businessType','address','zip','phone','email', 'contact','title','mode','initDate','lastDate','sessionConfirmed', 'dateTime','targetLocation','volunteer','notes','target','version']; sheet.appendRow(h); const ver = data.version || Date.now(); data.rows.forEach(r => sheet.appendRow([r.id,r.business,r.businessType||'', r.address,r.zip,r.phone,r.email,r.contact,r.title,r.mode,r.initDate, r.lastDate,r.sessionConfirmed,r.dateTime,r.targetLocation||'', r.volunteer,r.notes,data.target||21,ver])); } return ContentService.createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status:'error',msg:err.message})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { const cb = (e.parameter && e.parameter.callback) ? e.parameter.callback : 'cb'; try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet || sheet.getLastRow() < 2) return ContentService.createTextOutput(cb+'('+JSON.stringify({rows:[],target:21,version:0})+')') .setMimeType(ContentService.MimeType.JAVASCRIPT); const vals = sheet.getDataRange().getValues(); const headers = vals[0]; const rows = vals.slice(1).map(row => { const obj = {}; headers.forEach((h,i) => { obj[h] = (row[i]==null||row[i]===undefined)?'':String(row[i]); }); return obj; }); const target = rows.length ? (parseInt(rows[0].target)||21) : 21; const version = rows.length ? (parseInt(rows[0].version)||0) : 0; rows.forEach(r => { delete r.target; delete r.version; }); return ContentService.createTextOutput(cb+'('+JSON.stringify({rows,target,version})+')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } catch(err) { return ContentService.createTextOutput(cb+'('+JSON.stringify({rows:[],target:21,version:0,error:err.message})+')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } }
